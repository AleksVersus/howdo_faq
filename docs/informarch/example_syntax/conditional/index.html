Conditionals

single line conditional​

if x = 1: *pl '1' elseif x = 2: *pl '2' elseif x=3: *pl '3' else: *pl '4'
!@ Эта конструкция распознаётся, как часть многострочного условия.

if x = 1: *pl '1' elseif x = 2: *pl '2' elseif x=3: *pl '3' else:
! else подсвечивается как vaild, а должно как invalid       ^^^^
	*pl 'ddd'
end

elseif & ! elseif вне условия, invalid

! Простые примеры многострочных условий.

! многострочное условие без комментария после двоеточия:
if x > y and d > j:
	*pl 'Команды'
end

! многострочное условие с комментарием после двоеточия:
if x > y and d > j: ! валидный комментарий
	*pl 'Команды'
end

! многострочное условие с многострочным комментарием после двоеточия:
if x > y and d > j: ! валидный "
	многострочный " комментарий
	*pl 'Команды не закомментированы'
end

! Валидная структура многострочного условие с однострочными elseif и else:
if 0:
	elseif x = 1: *pl '1'
	elseif x = 2: *pl '2'
	elseif x = 3: *pl '3'
	elseif x = 4: *pl '4'
!   ^^^^^^ валидный elseif
	else: *pl '0'
!   ^^^^ валидный else с двоеточием
end

! Валидная структура многострочного условия с однострочными else if и else:
if 0:
	else if x = 1: *pl '1'
!   ^^^^^^^ валидный else if с раздельным написанием (эквивалент elseif)
	else if x = 2: *pl '2'
	else if x = 3: *pl '3'
	else if x = 4: *pl '4'
	else *pl '0'
!   ^^^^ валидный else без двоеточия
end

! лишний else
if 0:
	elseif x = 1: *pl '1'
	elseif x = 2: *pl '2'
	elseif x = 3: *pl '3'
	elseif x = 4: *pl '4'
	else: *pl '0'
!   ^^^^^ валидный else
	else: *pl '0'
!   ^^^^ невалидный else (дубликат)
end

! кривой else
if 0:
	elseif x = 1: *pl '1'
	elseif x = 2: *pl '2'
	elseif x = 3: *pl '3'
	elseif x = 4: *pl '4'
	else x = 5: *pl '0'
!   ^^^^^^^^^^^^^^^^^^^ вся строка невалидна, поскольку else не допускает условия
end

! кривой elseif
if 0:
	elseif x = 1: *pl '1'
	elseif x = 2: *pl '2'
	elseif x = 3: *pl '3'
	elseif x = 4: *pl '4'
	elseif: *pl '0'
!   ^^^^^^ невалидный elseif без условия
end

! кривой else if
if 0:
	else if x = 1: *pl '1'
!   ^^^^^^^ валидный else if с раздельным написанием
	else if x = 2: *pl '2'
	else if x = 3: *pl '3'
	else if x = 4: *pl '4'
	else if: *pl '5'
! ^^^^^^^^^^^^^^^^^^ вся строка невалидна из-за кривого else if
	else: *pl '0'
end

! лишний end в однострочном else
if 1:
	elseif x = 1: *pl '1'
	elseif x = 2: *pl '2'
	elseif x = 3: *pl '3'
	elseif x = 4: *pl '4'
	else: end
!         ^^^ невалидный end находится не на своём месте внутри пункта
end
! многострочные elseif и else, простые варианты:
if 0:
	! pass
elseif x = 1:
	*pl '1'
elseif x = 2:
	*pl '2'
elseif x = 3:
	*pl '3'
elseif x = 4:
	*pl '4'
else:
	*pl '0'
end

if 0:
	! pass
else if x = 1:
!^^^^^^ раздельное написание else if
	*pl '1'
else if x = 2:
	*pl '2'
else if x = 3:
	*pl '3'
else if x = 4:
	*pl '4'
else
!^^^ else без двоеточия
	*pl '0'
end

! многострочные elseif и else с комментариями после двоеточия
if 0:
	! pass
elseif x = 1: ! первый вариант
	*pl '1' & ! этот код будет выполняться, подсвечиваем, как код
elseif x = 2: ! второй вариант
	*pl '2'
elseif x = 3: ! третий вариант
	*pl '3'
elseif x = 4: ! четвёртый вариант
	*pl '4'
else: ! вариант по умолчанию
!     ^ начало комментария (else с ':' + хвост-комментарий)
	*pl '0'
!   ^^^^^^^ обычный код, потому что сразу после else написан валидный комментарий
!           а значит это не однострочный else.
end

! вариант с комментариями после else без двоеточия
if 0:
	! pass
elseif x = 1: ! первый вариант
	*pl '1' & ! этот код будет выполняться, подсвечиваем, как код
else ! вариант по умолчанию
!    ^ начало комментария (else без ':' + хвост-комментарий)
	*pl '0'
!   ^^^^^^^ обычный код, потому что сразу после else написан валидный комментарий
end

! однострочные elseif и else, хотя и с комментариями после двоеточия
if 0:
	! pass
elseif x = 1: & ! первый вариант
	*pl '1' & ! этот код НЕ будет выполняться, подсвечиваем, как комментарий
elseif x = 2: & ! второй вариант
	*pl '2'
elseif x = 3: & ! третий вариант
	*pl '3'
elseif x = 4: & ! четвёртый вариант
	*pl '4'
else: & ! вариант по умолчанию
	*pl '0'
!   ^^^^^^^ мёртвый код однострочного else = МНОГОСТРОЧНЫЙ комментарий
end & *pl 'fffff'

! МЁРТВЫЙ РЕГИОН однострочных else/elseif — ОДИН многострочный комментарий:
! структура у него ТА ЖЕ, что у обычного `!`-комментария: "…"/'…'/{…}.
! Поэтому плеер и подсветка одинаково понимают, где регион кончается:
! `if`/`end` ВНУТРИ {…} или незакрытого литерала регион НЕ закрывают —
! выйти можно только на строке, начинающейся с end/else/elseif/if.
if 0:
	! pass
elseif x = 1: & ! первый вариант — однострочный elseif
	*pl 'мёртвый код после однострочного elseif'
	$text = 'многострочный литерал
	end
	и его хвост'
	{ ! {…} — блочная структура комментария, считает вложенность
		'end'
		{ вложенный блок
			if 1:
				*pl 'и это всё ещё комментарий'
			end
		}
	}
	*pl 'после блока — всё ещё комментарий'
else: *pl '0'
	*pl 'мёртвый код после однострочного else'
	$s = "строка с end внутри
	end
	и хвост"
end
*pl 'после end — обычный код'

! вариант с else без двоеточия
if 0:
	! pass
elseif x = 1: & ! первый вариант
	*pl '1' & ! этот код НЕ будет выполняться, подсвечиваем, как комментарий
else & ! вариант по умолчанию
	*pl '0'
!   ^^^^^^^ мёртвый код однострочного else = МНОГОСТРОЧНЫЙ комментарий (до end)
end

! то же самое, что и выше, но вместо комментария после else обычный код
if 0:
	! pass
elseif x = 1: & ! первый вариант
	*pl '1' & ! этот код НЕ будет выполняться, подсвечиваем, как комментарий
else *pl '5'
	*pl '0'
!   ^^^^^^^ мёртвый код однострочного else = МНОГОСТРОЧНЫЙ комментарий (до end)
end

! ещё вариант, после else двоеточие:
if 0:
	!pass
elseif x = 1: *pl '1'
	gosub '1' & ! команды подсвечиваются, как комментарий
else: *pl '0'
	gosub '0' & ! не будет исполнена
end


if 1:
    dynamic {
        *pl 'код' & ! "}"
        *pl 'вторая строка'
    }
end
! тесты end`ов и вложенных условий

! end if — только `end` подсвечивается как ключевое слово, всё после него — комментарий:
if 1:
	!pass
end if

! вложенные условия
if 1:
	if 2:
		x = 3
	else:
		x = 4
	end
elseif x = 5: y = 9
else:
	if 0:
		new = old
	else: ! комментарий
		"Структура не ломается"
	end
end

! незакрытое условие руинит подсветку внутри фигурных скобок и/или подсветку конца локации
{
	if 1:
		else if x = 1: x
		else:
			gosub '1'
	! условие не закрыто
}
gosub '12' & ! остальные комманды подсвечиваются нормально

if [1, 2, 3] = [
	1, 
	2,
	3
]: *pl 'однострочное действие' else *pl 'работает'

if rand(1,
	23) = 12:
	*pl 'fkfkkfkfk'
else:
	*pl 'allalalaa'
end
! Всё подсвечивается корректно:
if ($a and $b):
    *pl "if baseline ok"
else:
    *pl "else baseline ok"
end

act ("desc", $x > 0):
    *pl "act baseline ok"
end

loop local $i while $i < 5 step $i += 1:
    *pl "<<$i>>"
end
! Многострочный if с многострочным условием
if (1 +
    2) = 3:
    *pl "тело if"
elseif (4 +
    5) = 9:
!^^^^^ elseif валиден (Фаза 7 починена)
    *pl "тело elseif"
else:
!^^^ else валиден (Фаза 7 починена)
    *pl "тело else"
end
!^^ end валиден (Фаза 7 починена)
! Однострочный if с многострочным условием
if [1,
    2,
    3] = $arr: *pl "массив в условии"
else:
!^^^ else подсвечивается как invalid — ВЕРНО: однострочная форма if (утверждение после ':')
    *pl "else"
end
!^^ end подсвечивается как invalid — ВЕРНО: однострочная форма
! Однострочная act c многострочным аргументом
act ("взять ключ",
    hp > 0): *pl "тело act"
end
!^^ end подсвечивается как invalid — ВЕРНО: однострочная форма act
! многострочная скобка внутри while многострочного loop
loop while (i <
    (5 +
    1)) step i += 1:
    *pl "тело loop <<i>>"
end
!^^ end валиден

! многострочность в loop ДО while (Фаза 8): tuple в инициализирующей группе,
! 'while' приходит только после закрытия скобки — заголовок валиден.
loop local i = [1,
    2,
    3 ] while i = [1,
    2,3] step o+=1:
!^^^ loop валиден (Фаза 8 починена: guard «while на строке» снят с bracket-arm)
    *pl 'ffff'
end
!^^ end валиден (блок открыт: 'while' наступил до ':')
loop local i = [1,
    2,
    3 ] step o+=1:
    *pl 'broken loop'
end
{
    loop local i = [1,
        2,
        3 ] while i = [1,2,3] step o = rand(1,
            23):
        *pl 'braced loop'
    end
}
if (a and
    b):
    act ("дело",
        c >
        d):
        *pl "внутри if, внутри act"
    end
    !^^ end валиден
end
!^^ end валиден
! Скобки внутри кавычек не ломают подсветку
! и закрывается на строке с ':'.
if ("текст (не скобка" +
    " ещё текст") = $s:
    *pl "проверка строковых скобок"
end
!^^ end валиден
!!! '!' после ':' на строке закрытия заголовка — не должен мешать открытию.
if (a and
    b): ! обычный комментарий после двоеточия
    !^^^^^^^^^^^^^ комментарий после ':' валиден
    *pl "тело"
end
!^^ end валиден
!!! 
{
    if (x and
        y):
        *pl "braced if тело"
    else:
    !^^^ else валиден
        *pl "braced else"
    end
    !^^ end валиден
}
